cmake_minimum_required(VERSION 3.20)

project(TezzNative C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

file(GLOB TEZZ_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
file(GLOB TEZZ_SPEECH_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/speech/*.c")

add_executable(tezzc ${TEZZ_SOURCES} ${TEZZ_SPEECH_SOURCES})
target_include_directories(tezzc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")

target_compile_definitions(tezzc PRIVATE TN_ENABLE_HOST_GUI=1)

# Native applications import this runtime instead of compiler-process addresses.
add_library(tnrt SHARED src/tnrt.c src/host_gui_backend.c src/security.c src/decimal.c runtime/native_alloc.c runtime/args.c runtime/gpu_bridge.c ${TEZZ_SPEECH_SOURCES})
target_include_directories(tnrt PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(tnrt PRIVATE TN_ENABLE_HOST_GUI=1 TN_NATIVE_X64_ABI=1)
set_target_properties(tnrt PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(NOT WIN32)
  set_target_properties(tnrt PROPERTIES OUTPUT_NAME tnrt-native)
else()
  set_target_properties(tnrt PROPERTIES PREFIX "")
endif()
add_dependencies(tezzc tnrt)

add_library(tzcpu SHARED lib/tzcpu_kernels.c)
target_compile_definitions(tzcpu PRIVATE TN_NATIVE_X64_ABI=1)
set_target_properties(tzcpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(NOT WIN32)
  set_target_properties(tzcpu PROPERTIES OUTPUT_NAME tzcpu-native)
  target_link_libraries(tzcpu PRIVATE m)
else()
  set_target_properties(tzcpu PROPERTIES PREFIX "")
endif()
add_dependencies(tezzc tzcpu)

option(TN_BUILD_CUDA_NATIVE "Build the CUDA native application library" OFF)
option(TN_NPU_OPENVINO "Enable dedicated Intel NPU model inference through OpenVINO" OFF)
option(TN_NPU_ORT_AMD "Enable dedicated AMD NPU model inference through a VitisAI plugin" OFF)
option(TN_NPU_ORT_QNN "Enable dedicated Qualcomm HTP model inference through a QNN plugin" OFF)
option(TN_NPU_HAILORT "Enable dedicated Hailo model inference through HailoRT" OFF)
set(npu_provider_count 0)
foreach(npu_option TN_NPU_OPENVINO TN_NPU_ORT_AMD TN_NPU_ORT_QNN TN_NPU_HAILORT)
  if(${npu_option})
    math(EXPR npu_provider_count "${npu_provider_count} + 1")
  endif()
endforeach()
if(npu_provider_count GREATER 1)
  message(FATAL_ERROR "Select exactly one dedicated NPU provider per runtime build")
endif()
if(TN_NPU_OPENVINO)
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  foreach(npu_target tezzc tnrt)
    target_compile_definitions(${npu_target} PRIVATE TN_NPU_OPENVINO=1)
    target_link_libraries(${npu_target} PRIVATE openvino::runtime::c)
  endforeach()
endif()

if(TN_NPU_ORT_AMD OR TN_NPU_ORT_QNN)
  set(TN_ORT_ROOT "" CACHE PATH "ONNX Runtime C SDK root (1.23 or later)")
  find_path(TN_ORT_INCLUDE_DIR onnxruntime_c_api.h HINTS "${TN_ORT_ROOT}/include" REQUIRED)
  find_library(TN_ORT_LIBRARY onnxruntime HINTS "${TN_ORT_ROOT}/lib" REQUIRED)
  foreach(npu_target tezzc tnrt)
    target_include_directories(${npu_target} SYSTEM PRIVATE "${TN_ORT_INCLUDE_DIR}")
    target_link_libraries(${npu_target} PRIVATE "${TN_ORT_LIBRARY}")
    if(TN_NPU_ORT_AMD)
      target_compile_definitions(${npu_target} PRIVATE TN_NPU_ORT_AMD=1)
    else()
      target_compile_definitions(${npu_target} PRIVATE TN_NPU_ORT_QNN=1)
    endif()
  endforeach()
endif()

if(TN_NPU_HAILORT)
  set(TN_HAILORT_ROOT "" CACHE PATH "HailoRT C SDK root")
  find_path(TN_HAILORT_INCLUDE_DIR hailo/hailort.h HINTS "${TN_HAILORT_ROOT}/include" REQUIRED)
  find_library(TN_HAILORT_LIBRARY hailort HINTS "${TN_HAILORT_ROOT}/lib" REQUIRED)
  foreach(npu_target tezzc tnrt)
    target_include_directories(${npu_target} SYSTEM PRIVATE "${TN_HAILORT_INCLUDE_DIR}")
    target_link_libraries(${npu_target} PRIVATE "${TN_HAILORT_LIBRARY}")
    target_compile_definitions(${npu_target} PRIVATE TN_NPU_HAILORT=1)
  endforeach()
endif()

if(TN_BUILD_CUDA_NATIVE)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(tzgpu SHARED lib/tzgpu_kernels.cu)
  target_compile_definitions(tzgpu PRIVATE TN_NATIVE_X64_ABI=1)
  target_link_libraries(tzgpu PRIVATE CUDA::cudart)
  set_target_properties(tzgpu PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
  if(NOT WIN32)
    set_target_properties(tzgpu PROPERTIES OUTPUT_NAME tzgpu-native)
  else()
    set_target_properties(tzgpu PROPERTIES PREFIX "")
  endif()
endif()

if(MSVC)
  target_compile_definitions(tezzc PRIVATE _CRT_SECURE_NO_WARNINGS)
  # /O2 = speed, /Oi = intrinsics, /Ot = fast, /GL = whole-program opt
  target_compile_options(tezzc PRIVATE /W4 /O2 /Oi /Ot /GL /GS- /fp:fast)
  target_link_options(tezzc PRIVATE /LTCG /OPT:REF /OPT:ICF)
else()
  target_compile_options(tezzc PRIVATE -O3 -Wall -Wextra -Wno-unused-parameter
    -Wno-unused-function -march=native -funroll-loops -fomit-frame-pointer)
  # LTO only on non-MinGW (MinGW ar requires -fuse-linker-plugin)
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug" AND NOT MINGW)
    target_compile_options(tezzc PRIVATE -flto)
    target_link_options(tezzc PRIVATE -flto)
  endif()
  if(MINGW)
    # MinGW-specific: enable LTO-safe optimizations
    target_compile_options(tezzc PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(tezzc PRIVATE -Wl,--gc-sections)
    target_link_options(tnrt PRIVATE -static-libgcc)
    target_link_options(tzcpu PRIVATE -static-libgcc)
  endif()
endif()

if(WIN32)
  target_link_libraries(tezzc PRIVATE ws2_32 secur32 crypt32 winhttp user32 gdi32 bcrypt)
  target_link_libraries(tnrt PRIVATE ws2_32 secur32 crypt32 winhttp user32 gdi32 bcrypt)
  target_link_libraries(tzcpu PRIVATE user32 gdi32 winmm)
  add_custom_command(TARGET tezzc POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:tezzc> "${CMAKE_CURRENT_SOURCE_DIR}/bin/tezzc-windows-x64.exe"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:tezzc> "${CMAKE_CURRENT_SOURCE_DIR}/bin/tezzc.exe")
else()
  find_package(OpenSSL REQUIRED)
  target_compile_definitions(tezzc PRIVATE TN_TLS_OPENSSL=1)
  target_compile_definitions(tnrt PRIVATE TN_TLS_OPENSSL=1)
  find_package(Threads REQUIRED)
  target_link_libraries(tezzc PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads m ${CMAKE_DL_LIBS})
  target_link_libraries(tnrt PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads m ${CMAKE_DL_LIBS})
  add_custom_command(TARGET tezzc POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:tezzc> "${CMAKE_CURRENT_SOURCE_DIR}/bin/tezzc-linux-x64")
endif()
